# 统一规则引擎设计

## 1. 问题分析

### 1.1 现有问题
当前规则引擎存在以下核心问题：

#### 1.1.1 规则分散，缺乏统一管理
- 规则分布在8个独立文件中：`base_rules.py`, `compose_rules.py`, `coroutine_rules.py`, `flow_rules.py`, `flow_lifecycle_rules.py`, `flow_structure_rules.py`, `hilt_rules.py`, `dagger2_rules.py`
- 每个文件独立开发，缺乏协调
- 新增规则需要修改多个地方：添加规则文件、更新import、修改review.py

#### 1.1.2 接口不一致，扩展性差
- 虽然大部分规则函数签名相同，但没有正式接口定义
- 难以添加新类型的规则（如AST分析、语义分析）
- 无法动态启用/禁用规则

#### 1.1.3 缺乏优先级和分类管理
- 严重级别（critical/major/minor）在规则内部硬编码
- 没有规则分类（安全、性能、最佳实践、可维护性等）
- 缺乏权重系统，所有规则扣分固定

#### 1.1.4 性能问题
- 所有规则串行执行
- 没有缓存机制，相同代码被多次分析
- 无法根据规则类型（CPU密集型/I/O密集型）优化执行顺序

### 1.2 影响分析
这些问题的具体影响包括：

1. **开发效率低下**：添加新规则需要了解整个代码结构
2. **维护困难**：规则逻辑分散，难以统一修改
3. **扩展性受限**：难以支持自定义规则
4. **用户体验差**：无法按需启用/禁用规则
5. **性能瓶颈**：大型代码库扫描时间长

## 2. 设计目标

### 2.1 核心目标
1. **统一管理**：集中式规则注册和管理
2. **灵活扩展**：支持动态添加/删除规则
3. **接口标准化**：定义统一的规则接口
4. **智能分类**：支持规则分类和优先级管理
5. **性能优化**：支持并行执行和缓存

### 2.2 非功能性目标
1. **向后兼容**：现有规则无需重写即可迁移
2. **高性能**：支持大型代码库的快速扫描
3. **可测试性**：便于单元测试和集成测试
4. **可配置性**：支持运行时配置调整

### 3.2 核心组件设计

#### 3.2.1 规则接口（Rule Interface）
```python
from abc import ABC, abstractmethod
from typing import Dict, List, Any, Optional
from dataclasses import dataclass
from enum import Enum

class RuleSeverity(Enum):
    """规则严重级别"""
    INFO = "info"      # 信息性提示
    MINOR = "minor"    # 轻微问题（扣5分）
    MAJOR = "major"    # 主要问题（扣10分）
    CRITICAL = "critical"  # 严重问题（扣20分）
    BLOCKER = "blocker"    # 阻塞性问题（扣100分）

class RuleCategory(Enum):
    """规则分类"""
    SECURITY = "security"          # 安全性
    PERFORMANCE = "performance"    # 性能
    BEST_PRACTICE = "best_practice" # 最佳实践
    MAINTAINABILITY = "maintainability" # 可维护性
    CORRECTNESS = "correctness"    # 正确性
    STYLE = "style"                # 代码风格

@dataclass
class RuleMetadata:
    """规则元数据"""
    id: str                      # 规则唯一标识符
    name: str                    # 规则名称（可读）
    description: str             # 规则描述
    severity: RuleSeverity       # 严重级别
    category: RuleCategory       # 分类
    enabled: bool = True         # 是否启用
    weight: float = 1.0          # 权重（影响评分）
    tags: List[str] = None       # 标签（如："android", "kotlin", "flow"）
    
    def __post_init__(self):
        if self.tags is None:
            self.tags = []

@dataclass
class Finding:
    """审查发现的问题"""
    rule_id: str                # 规则ID
    message: str                # 问题描述
    severity: RuleSeverity      # 严重级别
    file_path: Optional[str] = None  # 文件路径
    line_number: Optional[int] = None # 行号
    code_snippet: Optional[str] = None # 代码片段
    suggestion: Optional[str] = None # 修复建议

class Rule(ABC):
    """规则抽象基类"""
    
    @property
    @abstractmethod
    def metadata(self) -> RuleMetadata:
        """获取规则元数据"""
        pass
    
    @abstractmethod
    def check(self, context: 'RuleContext') -> List[Finding]:
        """
        检查代码，返回发现的问题
        
        Args:
            context: 规则上下文，包含代码、文件信息等
            
        Returns:
            发现的问题列表
        """
        pass
    
    def on_register(self):
        """规则注册时的回调（可选）"""
        pass
    
    def on_unregister(self):
        """规则注销时的回调（可选）"""
        pass
```

#### 3.2.3 规则注册表（Rule Registry）
```python
class RuleRegistry:
    """规则注册表，单例模式"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._rules = {}  # id -> Rule
            cls._instance._categories = {}  # category -> List[Rule]
            cls._instance._tags = {}  # tag -> List[Rule]
        return cls._instance
    
    def register(self, rule: Rule) -> None:
        """注册规则"""
        rule_id = rule.metadata.id
        
        if rule_id in self._rules:
            raise ValueError(f"规则ID已存在: {rule_id}")
        
        self._rules[rule_id] = rule
        rule.on_register()
        
        # 更新分类索引
        category = rule.metadata.category
        if category not in self._categories:
            self._categories[category] = []
        self._categories[category].append(rule)
        
        # 更新标签索引
        for tag in rule.metadata.tags:
            if tag not in self._tags:
                self._tags[tag] = []
            self._tags[tag].append(rule)
    
    def unregister(self, rule_id: str) -> None:
        """注销规则"""
        if rule_id not in self._rules:
            raise ValueError(f"规则ID不存在: {rule_id}")
        
        rule = self._rules[rule_id]
        rule.on_unregister()
        
        # 从分类索引移除
        category = rule.metadata.category
        self._categories[category].remove(rule)
        
        # 从标签索引移除
        for tag in rule.metadata.tags:
            self._tags[tag].remove(rule)
        
        del self._rules[rule_id]
    
    def get_rule(self, rule_id: str) -> Optional[Rule]:
        """根据ID获取规则"""
        return self._rules.get(rule_id)
    
    def get_rules_by_category(self, category: RuleCategory) -> List[Rule]:
        """根据分类获取规则"""
        return self._categories.get(category, [])
    
    def get_rules_by_tag(self, tag: str) -> List[Rule]:
        """根据标签获取规则"""
        return self._tags.get(tag, [])
    
    def get_all_rules(self, enabled_only: bool = True) -> List[Rule]:
        """获取所有规则"""
        rules = list(self._rules.values())
        if enabled_only:
            rules = [r for r in rules if r.metadata.enabled]
        return rules
    
    def enable_rule(self, rule_id: str):
        """启用规则"""
        rule = self.get_rule(rule_id)
        if rule:
            rule.metadata.enabled = True
    
    def disable_rule(self, rule_id: str):
        """禁用规则"""
        rule = self.get_rule(rule_id)
        if rule:
            rule.metadata.enabled = False
```

#### 3.2.4 规则执行器（Rule Engine）
```python
class RuleEngine:
    """规则执行引擎"""
    
    def __init__(self, registry: Optional[RuleRegistry] = None):
        self.registry = registry or RuleRegistry()
        self.cache = {}  # 简单内存缓存
    
    def execute_all(self, context: RuleContext) -> List[Finding]:
        """执行所有启用的规则"""
        findings = []
        rules = self.registry.get_all_rules(enabled_only=True)
        
        for rule in rules:
            rule_findings = self._execute_rule(rule, context)
            findings.extend(rule_findings)
        
        return findings
    
    def execute_by_category(self, context: RuleContext, category: RuleCategory) -> List[Finding]:
        """执行指定分类的规则"""
        findings = []
        rules = self.registry.get_rules_by_category(category)
        
        for rule in rules:
            if rule.metadata.enabled:
                rule_findings = self._execute_rule(rule, context)
                findings.extend(rule_findings)
        
        return findings
    
    def execute_parallel(self, context: RuleContext, max_workers: int = 4) -> List[Finding]:
        """并行执行规则（适用于CPU密集型规则）"""
        from concurrent.futures import ThreadPoolExecutor
        
        findings = []
        rules = self.registry.get_all_rules(enabled_only=True)
        
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            # 将规则分类：CPU密集型 vs I/O密集型
            cpu_intensive = []
            io_intensive = []
            
            for rule in rules:
                # 根据规则元数据或启发式方法分类
                if "ast" in rule.metadata.tags or "complex" in rule.metadata.tags:
                    cpu_intensive.append(rule)
                else:
                    io_intensive.append(rule)
            
            # 执行所有规则
            future_to_rule = {
                executor.submit(self._execute_rule, rule, context): rule
                for rule in rules
            }
            
            for future in concurrent.futures.as_completed(future_to_rule):
                rule_findings = future.result()
                findings.extend(rule_findings)
        
        return findings
    
    def _execute_rule(self, rule: Rule, context: RuleContext) -> List[Finding]:
        """执行单个规则"""
        try:
            # 检查缓存
            cache_key = f"{rule.metadata.id}:{hash(context.code)}"
            if cache_key in self.cache:
                return self.cache[cache_key]
            
            # 执行规则
            findings = rule.check(context)
            
            # 缓存结果
            self.cache[cache_key] = findings
            
            return findings
        except Exception as e:
            # 记录错误但不中断整个扫描
            error_finding = Finding(
                rule_id="rule_engine_error",
                message=f"规则执行失败 {rule.metadata.id}: {str(e)}",
                severity=RuleSeverity.MINOR,
                file_path=context.file_path
            )
            return [error_finding]
    
    def clear_cache(self):
        """清空缓存"""
        self.cache.clear()
```
